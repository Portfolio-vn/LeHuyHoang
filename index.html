01 // HARDWARE TEARDOWN

ESP32 IoT Air-Quality Station — Schematic & Fabrication

FLAGSHIP ENGINEERING BUILD · 2025–2026

Multi-Sensor Environmental Telemetry Station

ESP32-WROOM-32 Altium 2-Layer PCB 12-Bit ADC ThingSpeak REST

A fully independent hardware and firmware build engineered from basic circuit theory to complete PCB fabrication. Rather than using pre-packaged breakout modules, this station features a custom-routed power and signal conditioning board, empirical ppm calibration regression curves, local I2C OLED diagnostics, and an automatic WiFi telemetry pipeline pushing metrics to the cloud every 15 seconds.

⚙ 1. Sensor Conditioning & ADC

Three analog semiconductor gas sensors interfaced with ESP32 input-only GPIO lines:

  • MQ-7 (Carbon Monoxide): Precision voltage-divider circuit calibrated for 10–500 ppm range.
  • MQ-2 (Smoke & Hydrocarbons): Detects methane, butane, and smoke particles.
  • MQ-3 (Alcohol & VOC Vapor): Fast-response organic vapor telemetry.
  • 12-Bit ADC + Filtering: Dedicated low-pass RC filters & moving-average digital sampling to suppress high-frequency noise.

💾 2. Altium PCB Routing & Power

Custom printed circuit board engineered for high signal integrity and thermal stability:

  • Solid Ground Plane (GND): Star-ground topology isolating noisy 5V heater coils from sensitive analog lines.
  • Dual Power Regulation: Linear 5V rail for sensor heater coils (heating cycles) + high-efficiency 3.3V LDO for the ESP32 MCU and OLED.
  • Trace Impedance: Controlled trace routing for high-speed I2C clock (SCL) and data (SDA) lines.
  • Prototyping: Fabricated, hand-soldered, and verified on a digital storage oscilloscope.

📶 3. Firmware & Cloud Pipeline

Embedded C++ routines structured for fault-tolerant, uninterrupted field operation:

  • Empirical Calibration: Power-law regression models converting non-linear Rs/R0 ratios into verified ppm indices.
  • Local Diagnostics: Real-time SSD1306 OLED (128x64) graphical interface over hardware I2C bus.
  • ThingSpeak Cloud Push: Asynchronous HTTP REST requests transmitting telemetry packets every 15s.
  • Self-Healing Reconnect: Software watchdog and automated WiFi recovery logic if the AP connection drops.

FW_FLASH_01
REPO · LHHHSGS2022/esp32-iot-firmware · main.ino

Embedded C++ Firmware for the ESP32 Sensor Rig

C++ / Arduino-ESP32 3× MQ + DHT22 SSD1306 OLED · I2C

The complete firmware flashed onto the ESP32 rig. Every 2 seconds it samples three MQ gas sensors on the 12-bit ADC, converts each reading from the Rs/R0 resistance ratio to ppm with a power-law curve (ppm = a · (Rs/R0)b), reads temperature and humidity from a DHT22, then prints everything to a 128×64 OLED and the serial console.

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <DHT.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include "ThingSpeak.h"

// ================= OLED ==================
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET    -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// ================= MQ SENSOR PINS ==================
#define MQ5_PIN 32     // MQ-5 (LPG)
#define MQ3_PIN 33     // MQ-3 (Alcohol)
#define MQ2_PIN 35     // MQ-2 (Smoke)

// ================= DHT22 ==================
#define DHTPIN 25
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);

// ================= MQ Calibration ==================
float R0_MQ2 = 10.0;   // Change after calibration
float R0_MQ3 = 10.0;
float R0_MQ5 = 10.0;

float RL_VALUE = 10.0;  // Load resistor kΩ for MQ module

// MQ PPM FUNCTION
float MQ_GetPPM(int adc, float R0, float a, float b) {
  if (adc == 0) adc = 1;
  float RS = RL_VALUE * ( (4095.0 / adc) - 1.0 );
  float ratio = RS / R0;
  float ppm = a * pow(ratio, b);
  return ppm;
}

// ================= SETUP ==================
void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("\n===== ESP32 SENSOR SYSTEM BOOTING =====");

  // Init I2C (SDA=21, SCL=22)
  Serial.println("[I2C] Initializing I2C on SDA=21, SCL=22...");
  Wire.begin(27, 26);

  // Init OLED
  Serial.print("[OLED] Initializing OLED... ");
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("FAILED ❌");
    while (true);
  }
  Serial.println("OK ✔");

  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);

  // Init DHT22
  Serial.print("[DHT22] Initializing sensor... ");
  dht.begin();
  Serial.println("OK ✔");

  Serial.println("===== SYSTEM READY =====\n");
}


// ================= LOOP ==================
void loop() {

  // -------- READ MQ ANALOG --------
  Serial.println("------ Reading MQ Sensors ------");

  int mq5_adc = analogRead(MQ5_PIN);
  int mq3_adc = analogRead(MQ3_PIN);
  int mq2_adc = analogRead(MQ2_PIN);

  Serial.printf("[MQ5] ADC=%d\n", mq5_adc);
  Serial.printf("[MQ3] ADC=%d\n", mq3_adc);
  Serial.printf("[MQ2] ADC=%d\n", mq2_adc);

  // -------- CONVERT TO PPM --------

  // MQ2 – Smoke
  float mq2_ppm = MQ_GetPPM(mq2_adc, R0_MQ2, 114.75, -1.553);

  // MQ3 – Alcohol
  float mq3_ppm = MQ_GetPPM(mq3_adc, R0_MQ3, 4.8387, -3.318);

  // MQ5 – LPG
  float mq5_ppm = MQ_GetPPM(mq5_adc, R0_MQ5, 1000.5, -2.186);


  Serial.printf("[MQ2] Smoke = %.2f ppm\n", mq2_ppm);
  Serial.printf("[MQ3] Alcohol = %.2f ppm\n", mq3_ppm);
  Serial.printf("[MQ5] LPG = %.2f ppm\n", mq5_ppm);

  // -------- READ DHT22 --------
  Serial.println("------ Reading DHT22 ------");
  float humi = dht.readHumidity();
  float temp = dht.readTemperature();

  if (isnan(humi) || isnan(temp)) {
    Serial.println("[DHT22] ERROR ❌ Could not read data!");
  } else {
    Serial.printf("[DHT22] Temp: %.2f C | Humi: %.2f %%\n", temp, humi);
  }


  // -------- UPDATE OLED --------
  Serial.println("------ Updating OLED Display ------");

  display.clearDisplay();
  display.setCursor(0, 0);

  display.println(" ESP32 SENSOR DATA");
  display.println("----------------------");

  display.print("MQ2: ");
  display.print(mq2_ppm);
  display.println(" ppm");

  display.print("MQ3: ");
  display.print(mq3_ppm);
  display.println(" ppm");

  display.print("MQ5: ");
  display.print(mq5_ppm);
  display.println(" ppm");

  if (isnan(temp) || isnan(humi)) {
    display.println("DHT22 ERROR!");
  } else {
    display.print("T: ");
    display.print(temp);
    display.println(" C");

    display.print("H: ");
    display.print(humi);
    display.println(" %");
  }

  display.display();

  Serial.println("[OLED] Updated successfully ✔\n");

  delay(2000);
}
